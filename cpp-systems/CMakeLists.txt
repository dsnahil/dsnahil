cmake_minimum_required(VERSION 3.24)
project(CppSystems VERSION 1.0.0 LANGUAGES CXX)
include(CTest)
find_package(Threads REQUIRED)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(ENABLE_TSAN "Enable ThreadSanitizer (separate build)" OFF)
if(ENABLE_SANITIZERS AND ENABLE_TSAN)
  message(FATAL_ERROR "ASan and TSan require separate build directories")
endif()
add_library(project_options INTERFACE)
target_compile_options(project_options INTERFACE
  $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall;-Wextra;-Wpedantic;-Werror>
  $<$<CXX_COMPILER_ID:MSVC>:/W4;/WX;/permissive->)
if(ENABLE_SANITIZERS OR ENABLE_TSAN)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    message(FATAL_ERROR "Sanitizer configuration requires GCC or Clang")
  endif()
  if(ENABLE_TSAN)
    set(SANITIZERS thread)
  else()
    set(SANITIZERS address,undefined)
  endif()
  target_compile_options(project_options INTERFACE -fsanitize=${SANITIZERS} -fno-omit-frame-pointer)
  target_link_options(project_options INTERFACE -fsanitize=${SANITIZERS})
endif()
add_subdirectory(timingforge)
add_subdirectory(logicscope)
add_executable(systems_bench benchmarks/bench.cpp)
target_link_libraries(systems_bench PRIVATE timingforge_lib logicscope_lib)
if(BUILD_TESTING)
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py
    $<TARGET_FILE:timingforge> $<TARGET_FILE:logicscope>)
endif()
